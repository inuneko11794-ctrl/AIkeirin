​function parseFlexibleRaceText(rawText) {
const text = rawText.trim();
if (!text) return null;
const lines = text.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
let title = '即席コピペ診断レース';
let rawFormation = '';
let parsedLines = [];
const entryMap = {};
​lines.forEach((line, idx) => {
if (idx === 0 && !/^[1-9\s|｜]/.test(line) && !line.includes('[')) {
title = line.replace(/^[#【■●◆\s]+|[】\s]+$/g, '').slice(0, 28) || title;
}
const brackets = line.match(/[[^]]+]/g);
if (brackets && brackets.length >= 2) {
rawFormation = brackets.join(' ');
parsedLines = brackets.map(b => b.match(/\d/g)).filter(Boolean);
} else if (/並び|ライン|隊列/.test(line)) {
const rhs = line.split(/[：:|｜]/).slice(1).join(' ') || line;
const grps = rhs.split(/[\s-/・,]+/).map(g => g.match(/\d/g)).filter(Boolean);
if (grps.length > 0) {
rawFormation = grps.map(g => [${g.join(' ')}]).join(' ');
parsedLines = grps;
}

​function parseFormationString(rawStr, entries) {
if (!rawStr || !rawStr.trim()) return entries.map(e => [String(e.car)]);
const brackets = rawStr.match(/[[^]]+]/g);
if (brackets) return brackets.map(b => b.match(/\d/g)).filter(Boolean);
return rawStr.split(/[\s-/・,]+/).map(g => g.match(/\d/g)).filter(Boolean);
}
​function applyLineAndMarkRules(race) {
const carToEntry = {};
race.entries.forEach(e => { carToEntry[e.car] = e; });
const rawGroups = (race.lines && race.lines.length > 0) ? race.lines : race.entries.map(e => [String(e.car)]);
​let lineCounter = 0;
const lineSummaryList = [];
const carMetaMap = {};
​rawGroups.forEach(grp => {
const cars = grp.map(c => parseInt(c, 10)).filter(c => !isNaN(c) && carToEntry[c]);
if (cars.length === 0) return;
​if (cars.length === 1) {
const c = cars[0];
const rawM = normalizeRawMark(carToEntry[c].rawMark ?? carToEntry[c].mark);
carMetaMap[c] = { lineLabel: '単騎', isTanki: true, rawMark: rawM, convertedMark: 'T' };
lineSummaryList.push({ label: '単騎', isTanki: true, members: [{ car: c, name: carToEntry[c].name, rawMark: rawM, convertedMark: 'T' }] });
return;
}
​const label = ライン${CIRCLED_NUMS[lineCounter] || (lineCounter + 1)};
lineCounter++;
const rawMarks = cars.map(c => normalizeRawMark(carToEntry[c].rawMark ?? carToEntry[c].mark));
const convMarks = [...rawMarks];
​let i = 0;
while (i < rawMarks.length) {
if (rawMarks[i] === '無') {
let len = 1;
while (i + len < rawMarks.length && rawMarks[i + len] === '無') len++;
if (len >= 3) {
const seq = ['☆', '★', '星'];
for (let k = 0; k < len; k++) convMarks[i + k] = seq[k] || '星';
} else if (len === 2) {
convMarks[i] = '☆'; convMarks[i + 1] = '★';
}
i += len;
} else i++;
}
​for (let idx = 0; idx < rawMarks.length; idx++) {
if (rawMarks[idx] === '無' && convMarks[idx] === '無') {
const left = idx > 0 ? rawMarks[idx - 1] : '無';
const right = idx < rawMarks.length - 1 ? rawMarks[idx + 1] : '無';
const ref = left !== '無' ? left : right;
if (ref === '◎') convMarks[idx] = '⑳';
else if (ref === '◯') convMarks[idx] = '◯';
else if (ref === '▲' || ref === '△') convMarks[idx] = '▲';
else if (ref === '✕') convMarks[idx] = '✖';
else convMarks[idx] = '☆';
}
}

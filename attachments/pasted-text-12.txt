function extractFeatures(res) {
const f1 = Math.min(1.5, res.dissonance / (EMBEDDED_BASELINE.dissonanceMean * 1.5));
const f2 = Math.min(1.5, res.chaosIndex / (EMBEDDED_BASELINE.chaosMean * 1.8));
const f3 = 0.65;
const f4 = Math.min(1.5, (res.coupScore + res.abandonScore) / 4.5);
const f5 = Math.max(0, 1.0 - (res.gravityBias / (EMBEDDED_BASELINE.biasMean * 2.4)));
return [f1, f2, f3, f4, f5];
}
​function predictUpset(features) {
let z = aiModel.bias;
for (let i = 0; i < 5; i++) z += features[i] * aiModel.weights[i];
return 1 / (1 + Math.exp(-z));
}
